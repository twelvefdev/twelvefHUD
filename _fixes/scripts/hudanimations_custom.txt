event MenuOpen
{
	StopEvent	MenuClose 0.0 

	Animate HudMenu xpos "c-412"	 Accel 0 0

	// fade in
	Animate HudMenu Alpha 		"200"		Linear 0.0 0.2
	Animate HudMenu SelectionAlpha 	"200"		Linear 0.0 0.2
	Animate HudMenu MenuColor	"TanLight"	Linear 0.0 0.2
